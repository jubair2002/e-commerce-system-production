# 1. Update system & install Apache and PHP extensions
sudo apt update && sudo apt upgrade -y
sudo apt install -y \
  apache2 \
  libapache2-mod-php \
  php-cli \
  php-mysql \
  php-curl \
  php-mbstring \
  php-xml \
  php-zip \
  php-gd \
  php-intl \
  php-opcache

# 2. Enable Apache rewrite & headers modules
sudo a2enmod rewrite headers

# 3. Set path traversal permissions for /home/ubuntu
sudo chmod o+x /home/ubuntu
sudo chmod o+x /home/ubuntu/e-commerce-system-production

# 4. Create and configure external uploads directory
mkdir -p /home/ubuntu/files
sudo chown -R www-data:www-data /home/ubuntu/files
sudo chmod -R 775 /home/ubuntu/files

# 5. Create and configure internal storage & cache directory
mkdir -p /home/ubuntu/e-commerce-system-production/storage/cache
sudo chown -R www-data:www-data /home/ubuntu/e-commerce-system-production/storage
sudo chmod -R 775 /home/ubuntu/e-commerce-system-production/storage
sudo rm -rf /home/ubuntu/e-commerce-system-production/storage/cache/*

# 6. Run storage link
cd /home/ubuntu/e-commerce-system-production
php storage_link.php

# 7. Create Apache VirtualHost config
sudo bash -c 'cat << "EOF" > /etc/apache2/sites-available/ecommerce.conf
<VirtualHost *:80>
    ServerName _default_
    DocumentRoot /home/ubuntu/e-commerce-system-production

    <Directory /home/ubuntu/e-commerce-system-production>
        Options -Indexes +FollowSymLinks
        AllowOverride All
        Require all granted
        DirectoryIndex index.php index.html
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/ecommerce-error.log
    CustomLog ${APACHE_LOG_DIR}/ecommerce-access.log combined
</VirtualHost>
EOF'

# 8. Enable ecommerce site, test config, and restart Apache
sudo a2dissite 000-default.conf
sudo a2ensite ecommerce.conf
sudo apache2ctl configtest
sudo systemctl enable --now apache2
sudo systemctl restart apache2

# 9. Install background cron jobs directly under www-data user
sudo bash -c 'cat << "EOF" | crontab -u www-data -
* * * * * /usr/bin/php /home/ubuntu/e-commerce-system-production/cron/process_mail_queue.php >/dev/null 2>&1
* * * * * /usr/bin/php /home/ubuntu/e-commerce-system-production/cron/collect_metrics.php >/dev/null 2>&1
0 0 * * * /usr/bin/php /home/ubuntu/e-commerce-system-production/cron/cleanup_logs.php >/dev/null 2>&1
0 0 * * * /usr/bin/php /home/ubuntu/e-commerce-system-production/cron/customer_maintenance.php >/dev/null 2>&1
EOF'
